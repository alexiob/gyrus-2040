cmake_minimum_required(VERSION 3.20)
option(GYRUS_BUILD_FIRMWARE "Build RP2040 game and diagnostic firmware" OFF)
set(GYRUS_PRESENT_HZ "60" CACHE STRING "Device presentation rate (30 or60)")
if(NOT GYRUS_PRESENT_HZ MATCHES "^(30|60)$")
    message(FATAL_ERROR "GYRUS_PRESENT_HZ must be30 or60")
endif()
option(GYRUS_BUILD_DESKTOP "Build macOS Sokol desktop simulator" ON)
option(GYRUS_SANITIZERS "Enable host address/undefined sanitizers" OFF)
if(GYRUS_BUILD_FIRMWARE)
    set(PICO_BOARD waveshare_rp2040_touch_lcd_1.28 CACHE STRING "Board")
    if(NOT PICO_SDK_PATH)
        set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/.deps/pico-sdk")
    endif()
    # Before the SDK board header, whose LCD reset default is incorrect for our schematic.
    add_compile_definitions(WAVESHARE_LCD_RST_PIN=13)
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(gyrus2040 VERSION 0.2.0 LANGUAGES C CXX ASM)
if(GYRUS_BUILD_FIRMWARE)
    pico_sdk_init()
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(gyrus_diagnostic src/diagnostic.c)
target_include_directories(gyrus_diagnostic PUBLIC include)
target_compile_options(gyrus_diagnostic PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
add_library(gyrus_game_core src/game.c src/video.c src/render.c src/assets.c src/controls.c src/debug.c src/debug_game.c)
target_include_directories(gyrus_game_core PUBLIC include)
target_compile_options(gyrus_game_core PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
if(GYRUS_BUILD_FIRMWARE)
    add_executable(gyrus_game firmware/game_main.c firmware/display.c firmware/sensors.c firmware/buttons.c)
    target_include_directories(gyrus_game PRIVATE firmware)
    set_source_files_properties(firmware/game_main.c firmware/buttons.c
        PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-Wconversion;-Wshadow;-Wstrict-prototypes;-fstack-usage")
    target_link_libraries(gyrus_game PRIVATE gyrus_game_core gyrus_diagnostic pico_stdlib pico_multicore pico_flash pico_bootrom hardware_spi hardware_dma hardware_i2c hardware_pwm hardware_clocks hardware_watchdog)
    pico_enable_stdio_usb(gyrus_game 1)
    pico_enable_stdio_uart(gyrus_game 0)
    target_compile_definitions(gyrus_game PRIVATE GYRUS_PRESENT_HZ=${GYRUS_PRESENT_HZ} PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_ENABLE_IRQ_BACKGROUND_TASK=0 PICO_STACK_SIZE=0x2000 PICO_CORE1_STACK_SIZE=0x1000)
    pico_set_program_name(gyrus_game "gyrus_game")
    pico_set_program_version(gyrus_game "0.2.0")
    pico_add_linker_script_override_path(gyrus_game ${CMAKE_CURRENT_SOURCE_DIR}/firmware/linker FILES sections_stack.incl section_end.incl)
    pico_add_extra_outputs(gyrus_game)
    add_executable(gyrus_board_diag firmware/main.c firmware/display.c firmware/sensors.c)
    target_include_directories(gyrus_board_diag PRIVATE firmware)
    set_source_files_properties(firmware/main.c firmware/display.c firmware/sensors.c
        PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-Wconversion;-Wshadow;-Wstrict-prototypes;-fstack-usage")
    target_link_libraries(gyrus_board_diag PRIVATE gyrus_diagnostic pico_stdlib pico_bootrom pico_unique_id hardware_spi hardware_dma hardware_i2c hardware_pwm hardware_clocks hardware_adc hardware_flash)
    pico_enable_stdio_usb(gyrus_board_diag 1)
    pico_enable_stdio_uart(gyrus_board_diag 0)
    target_compile_definitions(gyrus_board_diag PRIVATE PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0)
    pico_set_program_name(gyrus_board_diag "gyrus_board_diag")
    pico_set_program_version(gyrus_board_diag "0.1.0")
    pico_add_extra_outputs(gyrus_board_diag)
else()
    enable_testing()
    add_executable(test_game tests/test_game.c)
    target_link_libraries(test_game PRIVATE gyrus_game_core)
    foreach(case initial route replay chance hits pause hazards bonus autofire)
        add_test(NAME game_${case} COMMAND test_game ${case})
    endforeach()
    foreach(name controls debug debug_game render video)
        add_executable(test_${name} tests/test_${name}.c)
        target_link_libraries(test_${name} PRIVATE gyrus_game_core)
        add_test(NAME ${name}_contract COMMAND test_${name})
    endforeach()
    add_executable(gyrus_headless host/headless.c)
    target_link_libraries(gyrus_headless PRIVATE gyrus_game_core)
    if(APPLE AND GYRUS_BUILD_DESKTOP)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sokol/sokol_app.h")
            message(FATAL_ERROR "Run python3 tools/setup.py for pinned Sokol headers, or set GYRUS_BUILD_DESKTOP=OFF")
        endif()
        enable_language(OBJC)
        add_executable(gyrus_desktop MACOSX_BUNDLE host/desktop.m)
        set_target_properties(gyrus_desktop PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "dev.iob.gyrus2040" MACOSX_BUNDLE_BUNDLE_NAME "Gyrus2040")
        target_include_directories(gyrus_desktop PRIVATE .deps/sokol)
        target_compile_options(gyrus_desktop PRIVATE -fobjc-arc)
        target_link_libraries(gyrus_desktop PRIVATE gyrus_game_core "-framework Cocoa" "-framework QuartzCore" "-framework Metal" "-framework MetalKit" "-framework AudioToolbox")
    endif()
    add_executable(test_diagnostic tests/test_diagnostic.c)
    target_link_libraries(test_diagnostic PRIVATE gyrus_diagnostic)
    target_compile_options(test_diagnostic PRIVATE -Wall -Wextra -Werror)
    if(GYRUS_SANITIZERS)
        foreach(t gyrus_diagnostic test_diagnostic gyrus_game_core test_game test_controls test_debug test_debug_game test_render test_video gyrus_headless)
            target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endforeach()
        foreach(t test_diagnostic test_game test_controls test_debug test_debug_game test_render test_video gyrus_headless)
            target_link_options(${t} PRIVATE -fsanitize=address,undefined)
        endforeach()
        if(TARGET gyrus_desktop)
            target_compile_options(gyrus_desktop PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(gyrus_desktop PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME diagnostic_contract COMMAND test_diagnostic)
    add_executable(test_display_fault tests/test_display_fault.c tests/fakes/fake_pico.c firmware/display.c)
    target_include_directories(test_display_fault PRIVATE tests/fakes firmware)
    target_link_libraries(test_display_fault PRIVATE gyrus_diagnostic)
    if(GYRUS_SANITIZERS)
        target_compile_options(test_display_fault PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test_display_fault PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME display_fault_contract COMMAND test_display_fault)
    add_executable(test_sensors tests/test_sensors.c firmware/sensors.c)
    target_include_directories(test_sensors PRIVATE tests/sensor_fakes firmware)
    target_link_libraries(test_sensors PRIVATE gyrus_diagnostic)
    target_compile_options(test_sensors PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -UNDEBUG)
    if(GYRUS_SANITIZERS)
        target_compile_options(test_sensors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test_sensors PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME sensors_contract COMMAND test_sensors)
    foreach(t test_diagnostic test_display_fault test_game test_controls test_debug test_debug_game test_render test_video)
        target_compile_options(${t} PRIVATE -UNDEBUG)
    endforeach()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME serial_line_contract COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_serial_lines.py)
    add_test(NAME debug_client_contract COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_gyrusctl.py)
    set_tests_properties(debug_client_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(serial_line_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
